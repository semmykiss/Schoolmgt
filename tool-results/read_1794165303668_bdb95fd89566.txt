    18→function mulberry32(seed: number) {
    19→  let a = seed
    20→  return () => {
    21→    a |= 0; a = (a + 0x6d2b79f5) | 0
    22→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    23→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    24→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    25→  }
    26→}

    35→function lastWeekdays(count: number): string[] {
    36→  const out: string[] = []
    37→  const d = new Date()
    38→  while (out.length < count) {
    39→    const dow = d.getDay()
    40→    if (dow !== 0 && dow !== 6) out.push(iso(d))
    41→    d.setDate(d.getDate() - 1)
    42→  }
    43→  return out.reverse()
    44→}
    45→function nextWeekdays(count: number, skipToday = true): string[] {
    46→  const out: string[] = []
    47→  const d = addDays(new Date(), skipToday ? 1 : 0)
    48→  while (out.length < count) {
    49→    const dow = d.getDay()
    50→    if (dow !== 0 && dow !== 6) out.push(iso(d))
    51→    d.setDate(d.getDate() + 1)
    52→  }
    53→  return out
    54→}
