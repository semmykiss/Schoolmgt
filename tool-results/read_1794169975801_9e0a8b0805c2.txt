   111→  })
   112→  const classes = classesData?.items ?? []
   113→  const form = useForm<CreateValues>({
   114→    resolver: zodResolver(createSchema),
   115→    defaultValues: { firstName: '', lastName: '', otherName: '', gender: 'MALE', dateOfBirth: '', classId: 'NONE', sectionId: 'NONE', admissionDate: new Date().toISOString().slice(0, 10) },

   117→  const classId = form.watch('classId')
   118→  const sections = classes.find(c => c.id === classId)?.sections ?? []
   119→  const errs = form.formState.errors
   120→
   121→  const create = useMutation({
   122→    mutationFn: (v: CreateValues) => api<{ admissionNumber: string }>('/students', {
   123→      method: 'POST',
   124→      body: {
   125→        ...v,
   126→        otherName: v.otherName || undefined,
   127→        dateOfBirth: v.dateOfBirth || null,
   128→        classId: v.classId === 'NONE' ? null : v.classId,

   228→// ── import dialog ───────────────────────────────────────────────────────────
   229→interface ImportReport { row: number; status: string; admissionNumber?: string; message?: string }
   230→
   231→function ImportDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   232→  const qc = useQueryClient()
   233→  const [fileName, setFileName] = useState('')
   234→  const [preview, setPreview] = useState<Record<string, string>[]>([])
   235→  const [report, setReport] = useState<ImportReport[] | null>(null)
   236→
   237→  const onFile = async (file: File) => {
   238→    const text = await file.text()
   239→    const rows = parseCsv(text)
   240→    if (rows.length < 2) { toast.error('The CSV needs a header row and at least one student'); return }
   241→    const headers = rows[0].map(h => h.trim().toLowerCase())
   242→    const wanted = ['first name', 'last name', 'gender', 'class', 'date of birth', 'guardian name', 'guardian phone']
   243→    const missing = wanted.filter(w => !headers.includes(w))
   244→    if (missing.length) { toast.error(`Missing column(s): ${missing.join(', ')}`); return }
   245→    setFileName(file.name)
   246→    setPreview(rows.slice(1).map(r => Object.fromEntries(headers.map((h, i) => [h, (r[i] ?? '').trim()]))))
   247→    setReport(null)
   248→  }
   249→
   250→  const doImport = useMutation({
   251→    mutationFn: () => api<{ results: ImportReport[]; created: number; failed: number }>('/students/import', {
   252→      method: 'POST',
   253→      body: {
   254→        rows: preview.map(r => ({
   255→          firstName: r['first name'], lastName: r['last name'],
   256→          gender: (r['gender'] || 'MALE').toUpperCase() === 'FEMALE' ? 'FEMALE' : 'MALE',
   257→          className: r['class'] || '',
   258→          dateOfBirth: r['date of birth'] || '',
   259→          guardianName: r['guardian name'] || '',
   260→          guardianPhone: r['guardian phone'] || '',
   261→        })),
   262→      },
   263→    }),
   264→    onSuccess: r => {
   265→      setReport(r.results)
   266→      qc.invalidateQueries({ queryKey: ['students'] })
   267→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   268→      toast.success(`${r.created} student(s) imported${r.failed ? `, ${r.failed} failed` : ''}`)
   269→    },
   270→    onError: err => toast.error(err instanceof Error ? err.message : 'Import failed'),
   271→  })
   272→
   273→  const reset = () => { setPreview([]); setFileName(''); setReport(null) }
   274→
   275→  return (
   276→    <Dialog open={open} onOpenChange={o => { if (!o) { reset(); onOpenChange(false) } }}>
   277→      <DialogContent className="sm:max-w-lg">
   278→        <DialogHeader>
   279→          <DialogTitle>Import students from CSV</DialogTitle>
   280→          <DialogDescription>
   281→            Columns: <code className="text-xs">First Name, Last Name, Gender, Class, Date of Birth, Guardian Name, Guardian Phone</code>
   282→          </DialogDescription>
   283→        </DialogHeader>
   284→        {report ? (
   285→          <div className="max-h-72 space-y-1 overflow-y-auto rounded-lg border p-3 [scrollbar-width:thin]">
   286→            {report.map(r => (
   287→              <p key={r.row} className="text-sm">
   288→                <span className="text-muted-foreground">Row {r.row}:</span>{' '}
   289→                {r.status === 'CREATED'
   290→                  ? <span className="text-emerald-600 dark:text-emerald-400">created {r.admissionNumber}</span>
   291→                  : <span className="text-destructive">failed — {r.message}</span>}
   292→              </p>
   293→            ))}
   294→          </div>
   295→        ) : preview.length === 0 ? (
   296→          <label className="flex cursor-pointer flex-col items-center justify-center gap-2 rounded-xl border-2 border-dashed p-8 text-center transition-colors hover:border-primary/50 hover:bg-muted/40">
   297→            <FileUp className="h-8 w-8 text-muted-foreground" aria-hidden />
   298→            <span className="text-sm font-medium">{fileName || 'Choose a .csv file'}</span>
   299→            <span className="text-xs text-muted-foreground">Exported from Excel, Google Sheets or another school software</span>
   300→            <input type="file" accept=".csv,text/csv" className="sr-only" onChange={e => { const f = e.target.files?.[0]; if (f) void onFile(f) }} />
   301→          </label>
   302→        ) : (
   303→          <div className="space-y-3">
   304→            <p className="text-sm"><span className="font-medium">{preview.length} row(s)</span> ready to import.</p>
   305→            <div className="max-h-40 overflow-y-auto rounded-lg border [scrollbar-width:thin]">
   306→              <table className="w-full text-xs">
   307→                <thead className="bg-muted/60"><tr><th className="p-2 text-left">Name</th><th className="p-2 text-left">Gender</th><th className="p-2 text-left">Class</th></tr></thead>
   308→                <tbody>
   309→                  {preview.slice(0, 8).map((r, i) => (
   310→                    <tr key={i} className="border-t"><td className="p-2">{r['first name']} {r['last name']}</td><td className="p-2">{r['gender']}</td><td className="p-2">{r['class'] || '—'}</td></tr>
   311→                  ))}
   312→                </tbody>
   313→              </table>
   314→            </div>
   315→          </div>
   316→        )}
   317→        <DialogFooter>
   318→          {report ? (
   319→            <Button onClick={() => { reset(); onOpenChange(false) }}>Done</Button>
   320→          ) : preview.length > 0 ? (
   321→            <>
   322→              <Button variant="ghost" onClick={reset}>Pick another file</Button>
   323→              <Button onClick={() => doImport.mutate()} disabled={doImport.isPending}>
   324→                {doImport.isPending && <Loader2 className="animate-spin" aria-hidden />}
   325→                Import {preview.length} students
   326→              </Button>
   327→            </>
   328→          ) : (
   329→            <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   330→          )}
   331→        </DialogFooter>
   332→      </DialogContent>
   333→    </Dialog>
   334→  )
   335→}

   846→  const [exitTarget, setExitTarget] = useState<StudentDetail | null>(null)
   847→  const [selectedId, setSelectedId] = useState<string | null>(null)
   848→
   849→  const { data: classesData } = useQuery({
   850→    queryKey: qk.classes(),
   851→    queryFn: () => api<{ items: { id: string; name: string; sections: { id: string; name: string }[] }[] }>('/academics/classes'),
   852→  })
   853→  const classes = classesData?.items ?? []
   854→
   855→  const filters = { classId: classFilter, status: statusFilter }
   856→  const { data, isLoading } = useQuery({
   857→    queryKey: qk.students(filters),
   858→    queryFn: () => api<{ items: StudentListRow[]; counts: Record<string, number> }>('/students', {
   859→      query: { classId: classFilter === 'ALL' ? undefined : classFilter, status: statusFilter === 'ALL' ? undefined : statusFilter },
   860→    }),
   861→  })
   862→  const students = data?.items ?? []
   863→  const counts = data?.counts ?? {}
   864→
   865→  const removeStudent = useMutation({
   866→    mutationFn: (id: string) => api(`/students/${id}`, { method: 'DELETE' }),
   867→    onSuccess: () => {
   868→      qc.invalidateQueries({ queryKey: ['students'] })
   869→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   870→      toast.success('Student deleted')
   871→    },
   872→    onError: err => toast.error(err instanceof Error ? err.message : 'Delete failed'),
   873→  })
   874→
   875→  const columns = useMemo<ColumnDef<StudentListRow, unknown>[]>(() => [
   876→    {
   877→      accessorKey: 'lastName',
   878→      header: 'Student',
   879→      meta: { label: 'Student' },
   880→      cell: ({ row }) => (
   881→        <div>
   882→          <p className="font-medium leading-tight">{row.original.lastName} {row.original.firstName}{row.original.otherName ? ` (${row.original.otherName})` : ''}</p>
   883→          <p className="font-mono text-xs text-muted-foreground">{row.original.admissionNumber}</p>
   884→        </div>
   885→      ),
   886→    },
   887→    {
   888→      id: 'class',
   889→      accessorFn: r => r.class?.name ?? '',
   890→      header: 'Class',
   891→      meta: { label: 'Class' },
   892→      cell: ({ row }) => row.original.class
   893→        ? <Badge variant="outline">{row.original.class.name}{row.original.section ? ` ${row.original.section.name}` : ''}</Badge>
   894→        : <span className="text-muted-foreground">—</span>,
   895→    },
   896→    { accessorKey: 'gender', header: 'Gender', meta: { label: 'Gender' }, cell: ({ getValue }) => <span className="text-sm text-muted-foreground">{String(getValue())}</span> },
   897→    {
   898→      accessorKey: 'dateOfBirth',
   899→      header: 'Born',
   900→      meta: { label: 'Born' },
   901→      cell: ({ row }) => <span className="text-sm text-muted-foreground">{row.original.dateOfBirth ? fmtDate(row.original.dateOfBirth) : '—'}</span>,
   902→    },
   903→    {
   904→      accessorKey: 'status',
   905→      header: 'Status',
   906→      meta: { label: 'Status' },
   907→      cell: ({ row }) => {
   908→        const meta = STATUS_META[row.original.status]
   909→        return meta ? <Badge variant="outline" className={meta.chip}>{meta.label}</Badge> : row.original.status
   910→      },
   911→    },
   912→    {
   913→      id: 'actions',
   914→      header: '',
   915→      enableSorting: false, enableHiding: false,
   916→      meta: { label: 'Actions' },
   917→      cell: ({ row }) => {
   918→        const s = row.original
   919→        if (!canWrite) return null
   920→        return (
   921→          <DropdownMenu>
   922→            <DropdownMenuTrigger asChild>
   923→              <Button variant="ghost" size="icon" className="h-8 w-8" aria-label={`Actions for ${s.firstName}`} onClick={e => e.stopPropagation()}>
   924→                <MoreHorizontal className="h-4 w-4" />
   925→              </Button>
   926→            </DropdownMenuTrigger>
   927→            <DropdownMenuContent align="end" className="w-48" onClick={e => e.stopPropagation()}>
   928→              {s.status === 'ACTIVE' && (
   929→                <DropdownMenuItem onSelect={async () => {
   930→                  const detail = await api<StudentDetail>(`/students/${s.id}`)
   931→                  setExitTarget(detail)
   932→                }}>
   933→                  <DoorOpen aria-hidden /> Exit school…
   934→                </DropdownMenuItem>
   935→              )}
   936→              <DropdownMenuSeparator />
   937→              <DropdownMenuItem variant="destructive" onSelect={() => setDeleteTarget(s)}>
   938→                <Trash2 aria-hidden /> Delete
   939→              </DropdownMenuItem>
   940→            </DropdownMenuContent>
   941→          </DropdownMenu>
   942→        )
   943→      },
   944→    },
   945→  ], [canWrite])
   946→
   947→  const exportUrl = useMemo(() => {
   948→    const params = new URLSearchParams()
   949→    if (classFilter !== 'ALL') params.set('classId', classFilter)
   950→    if (statusFilter !== 'ALL') params.set('status', statusFilter)
   951→    const qs = params.toString()
   952→    return `/api/v1/students/export${qs ? `?${qs}` : ''}`
   953→  }, [classFilter, statusFilter])
   954→
   955→  // detail panel mode
   956→  if (selectedId) {
   957→    return (
   958→      <div className="space-y-4">
   959→        <StudentDetailPanel studentId={selectedId} onBack={() => setSelectedId(null)} />
   960→        {exitTarget && <ExitDialog student={exitTarget} onClose={() => setExitTarget(null)} />}
   961→      </div>
   962→    )
   963→  }
   964→
   965→  return (
   966→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.25 }} className="space-y-6">
   967→      <div className="flex flex-wrap gap-2">
   968→        {['ACTIVE', 'GRADUATED', 'TRANSFERRED', 'SUSPENDED', 'LEFT'].map(st => {
   969→          const active = statusFilter === st
   970→          const meta = STATUS_META[st]
   971→          return (
   972→            <button key={st} type="button" onClick={() => setStatusFilter(active ? 'ALL' : st)} aria-pressed={active}
   973→              className={`rounded-full border px-3.5 py-1.5 text-xs font-medium transition-colors ${active ? 'border-primary bg-primary/10 text-primary' : 'border-border text-muted-foreground hover:bg-muted/60'}`}>
   974→              {meta.label} · {counts[st] ?? 0}
   975→            </button>
   976→          )
   977→        })}
   978→      </div>
   979→
   980→      <DataTable<StudentListRow>
   981→        columns={columns}
   982→        data={students}
   983→        loading={isLoading}
   984→        rowId={s => s.id}
   985→        onRowClick={s => setSelectedId(s.id)}
   986→        searchPlaceholder="Search name or admission number…"
   987→        searchExtractor={s => `${s.firstName} ${s.lastName} ${s.otherName ?? ''} ${s.admissionNumber}`}
   988→        initialPageSize={15}
   989→        toolbar={
   990→          <div className="flex flex-wrap items-center gap-2">
   991→            <Select value={classFilter} onValueChange={setClassFilter}>
   992→              <SelectTrigger size="sm" className="w-40" aria-label="Filter by class">
   993→                <SelectValue />
   994→              </SelectTrigger>
   995→              <SelectContent>
   996→                <SelectItem value="ALL">All classes</SelectItem>
   997→                {classes.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   998→              </SelectContent>
   999→            </Select>
  1000→            <Button size="sm" variant="outline" asChild>
  1001→              <a href={exportUrl} download><Download aria-hidden /> Export CSV</a>
  1002→            </Button>
  1003→            {canWrite && (
  1004→              <>
  1005→                <Button size="sm" variant="outline" onClick={() => setImportOpen(true)}><FileUp aria-hidden /> Import CSV</Button>
  1006→                <Button size="sm" onClick={() => setCreateOpen(true)}><Plus aria-hidden /> Enrol student</Button>
  1007→              </>
  1008→            )}
  1009→          </div>
  1010→        }
  1011→        emptyState={
  1012→          <EmptyState
  1013→            icon={Users}
  1014→            title="No students here"
  1015→            description={statusFilter !== 'ALL' || classFilter !== 'ALL' ? 'Try clearing the filters.' : 'Enrol your first student or import a CSV from your previous system.'}
  1016→            action={canWrite ? <Button onClick={() => setCreateOpen(true)}><Plus aria-hidden /> Enrol student</Button> : undefined}
  1017→          />
  1018→        }
  1019→      />
  1020→
  1021→      <CreateStudentDialog open={createOpen} onOpenChange={setCreateOpen} />
  1022→      <ImportDialog open={importOpen} onOpenChange={setImportOpen} />
  1023→      <ConfirmDialog
  1024→        open={deleteTarget !== null}
  1025→        onOpenChange={o => { if (!o) setDeleteTarget(null) }}
  1026→        title={`Delete ${deleteTarget?.firstName} ${deleteTarget?.lastName}?`}
  1027→        description="Students with invoices or exam results cannot be deleted — record an exit instead."
  1028→        confirmLabel="Delete student" destructive
  1029→        onConfirm={async () => { if (deleteTarget) await removeStudent.mutateAsync(deleteTarget.id) }}
  1030→      />
  1031→    </motion.div>
  1032→  )
  1033→}
