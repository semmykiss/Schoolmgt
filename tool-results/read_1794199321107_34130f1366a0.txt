    33→interface ExamRow {

    47→const formSchema = z.object({
    48→  name: z.string().min(3, 'Give the exam a name (min 3 characters)').max(140),
    49→  termId: z.string().min(1, 'Select the term'),
    50→  examTypeId: z.string().min(1, 'Select the exam type'),
    51→  startDate: z.string().optional(),
    52→  endDate: z.string().optional(),
    53→}).refine(v => !v.startDate || !v.endDate || v.startDate <= v.endDate, {
    54→  message: 'End date cannot be before the start date',
    55→  path: ['endDate'],
    56→})

   225→  return (

   296→// ── Create / edit dialog ────────────────────────────────────────────────────
   297→function ExamDialog({ open, onOpenChange, editTarget, examTypes, terms, onDone }: {
   298→  open: boolean
   299→  onOpenChange: (o: boolean) => void
   300→  editTarget: ExamRow | null
   301→  examTypes: ExamTypeRow[]
   302→  terms: (SessionRow['terms'][number] & { sessionName: string })[]
   303→  onDone: () => void
   304→}) {
   305→  const qc = useQueryClient()
   306→  const form = useForm<FormValues>({
   307→    resolver: zodResolver(formSchema),
   308→    defaultValues: { name: '', termId: '', examTypeId: '', startDate: '', endDate: '' },
   309→  })
   310→
   311→  // Re-seed on open
   312→  const [seededKey, setSeededKey] = useState('')
   313→  const seedKey = `${open}:${editTarget?.id ?? 'new'}`
   314→  if (open && seededKey !== seedKey) {
   315→    setSeededKey(seedKey)
   316→    form.reset(editTarget ? {
   317→      name: editTarget.name,
   318→      termId: editTarget.term.id,
   319→      examTypeId: editTarget.examType.id,
   320→      startDate: editTarget.startDate ?? '',
   321→      endDate: editTarget.endDate ?? '',
   322→    } : { name: '', termId: '', examTypeId: '', startDate: '', endDate: '' })
   323→  }
   324→
   325→  const save = useMutation({
   326→    mutationFn: (v: FormValues) => {
   327→      const body = {
   328→        name: v.name,
   329→        termId: v.termId,
   330→        examTypeId: v.examTypeId,
   331→        startDate: v.startDate || '',
   332→        endDate: v.endDate || '',
   333→      }
   334→      return editTarget
   335→        ? api(`/exams/${editTarget.id}`, { method: 'PATCH', body })
   336→        : api('/exams', { method: 'POST', body })
   337→    },
   338→    onSuccess: exam => {
   339→      qc.invalidateQueries({ queryKey: ['exams'] })
   340→      onDone()
   341→      onOpenChange(false)
   342→      toast.success(editTarget ? 'Exam updated' : `Exam created — ${(exam as ExamRow).name}`)
   343→    },
   344→    onError: e => toast.error(e instanceof Error ? e.message : 'Save failed'),
   345→  })
   346→
   347→  const errs = form.formState.errors
   348→
   349→  return (
   350→    <Dialog open={open} onOpenChange={o => { onOpenChange(o); if (!o) form.reset() }}>
   351→      <DialogContent className="sm:max-w-md">
   352→        <DialogHeader>
   353→          <DialogTitle>{editTarget ? 'Edit exam' : 'New exam'}</DialogTitle>
   354→          <DialogDescription>
   355→            {editTarget ? 'Update the exam details.' : 'One exam of each type per term — e.g. a Mid-Term Test and a Final Exam.'}
   356→          </DialogDescription>
   357→        </DialogHeader>
   358→        <form className="space-y-4" onSubmit={form.handleSubmit(v => save.mutate(v))} noValidate>
   359→          <div className="space-y-1.5">
   360→            <Label htmlFor="exam-name">Name</Label>
   361→            <Input id="exam-name" placeholder="First Term Examination — 2024/2025" {...form.register('name')} aria-invalid={!!errs.name} />
   362→            <FieldError message={errs.name?.message} />
   363→          </div>
   364→          <div className="grid gap-3 sm:grid-cols-2">
   365→            <div className="space-y-1.5">
   366→              <Label>Term</Label>
   367→              <Select value={form.watch('termId')} onValueChange={v => form.setValue('termId', v, { shouldValidate: true })}>
   368→                <SelectTrigger aria-label="Select term" aria-invalid={!!errs.termId}><SelectValue placeholder="Select term" /></SelectTrigger>
   369→                <SelectContent>
   370→                  {terms.map(t => (
   371→                    <SelectItem key={t.id} value={t.id}>
   372→                      {TERM_LABEL[t.name] ?? t.name} — {t.sessionName}{t.isCurrent ? ' (current)' : ''}
   373→                    </SelectItem>
   374→                  ))}
   375→                </SelectContent>
   376→              </Select>
   377→              <FieldError message={errs.termId?.message} />
   378→            </div>
   379→            <div className="space-y-1.5">
   380→              <Label>Type</Label>
   381→              <Select value={form.watch('examTypeId')} onValueChange={v => form.setValue('examTypeId', v, { shouldValidate: true })}>
   382→                <SelectTrigger aria-label="Select exam type" aria-invalid={!!errs.examTypeId}><SelectValue placeholder="Select type" /></SelectTrigger>
   383→                <SelectContent>
   384→                  {examTypes.map(t => <SelectItem key={t.id} value={t.id}>{t.name} ({t.weight}%)</SelectItem>)}
   385→                </SelectContent>
   386→              </Select>
   387→              <FieldError message={errs.examTypeId?.message} />
   388→            </div>
   389→          </div>
   390→          <div className="grid gap-3 sm:grid-cols-2">
   391→            <div className="space-y-1.5">
   392→              <Label htmlFor="exam-start">Starts (optional)</Label>
   393→              <Input id="exam-start" type="date" {...form.register('startDate')} />
   394→            </div>
   395→            <div className="space-y-1.5">
   396→              <Label htmlFor="exam-end">Ends (optional)</Label>
   397→              <Input id="exam-end" type="date" {...form.register('endDate')} />
   398→              <FieldError message={errs.endDate?.message} />
   399→            </div>
   400→          </div>
   401→          <DialogFooter>
   402→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   403→            <Button type="submit" disabled={save.isPending}>
   404→              {save.isPending && <Loader2 className="h-4 w-4 animate-spin" aria-hidden />}
   405→              {editTarget ? 'Save changes' : 'Create exam'}
   406→            </Button>
   407→          </DialogFooter>
   408→        </form>
   409→      </DialogContent>
   410→    </Dialog>
   411→  )
   412→}

   415→interface ScheduleRow {
   416→  id: string
   417→  date: string
   418→  startTime: string
   419→  endTime: string
   420→  maxMarks: number
   421→  passMarks: number
   422→  classSubject: {
   423→    id: string
   424→    subject: { name: string; code: string | null }
   425→    class: { id: string; name: string }
   426→    teacher: { id: string; firstName: string; lastName: string } | null
   427→  }
   428→}
   429→interface ClassOption { id: string; name: string; level: number }
   430→interface ClassSubjectOption {
   431→  id: string
   432→  subject: { name: string; code: string | null }
   433→  teacher: { fullName: string } | null

   436→function PapersDialog({ exam, open, onOpenChange, canWrite, onChanged }: {
   437→  exam: ExamRow | null
   438→  open: boolean
   439→  onOpenChange: (o: boolean) => void
   440→  canWrite: boolean
   441→  onChanged: () => void
   442→}) {
   443→  const qc = useQueryClient()
   444→  const examId = exam?.id ?? ''
   445→  const [formOpen, setFormOpen] = useState(false)
   446→  const [editTarget, setEditTarget] = useState<ScheduleRow | null>(null)
   447→  const [deleteTarget, setDeleteTarget] = useState<ScheduleRow | null>(null)
   448→  const [classId, setClassId] = useState('NONE')
   449→  const [classSubjectId, setClassSubjectId] = useState('NONE')
   450→  const [date, setDate] = useState('')
   451→  const [startTime, setStartTime] = useState('09:00')
   452→  const [endTime, setEndTime] = useState('11:00')
   453→  const [maxMarks, setMaxMarks] = useState('100')
   454→  const [passMarks, setPassMarks] = useState('40')
   455→  const [formError, setFormError] = useState('')
   456→
   457→  const { data, isLoading } = useQuery({
   458→    queryKey: qk.examSchedules(open && examId ? examId : undefined),
   459→    queryFn: () => api<{ items: ScheduleRow[] }>('/exams/schedules', { query: { examId } }),
   460→    enabled: open && !!examId,
   461→  })
   462→  const schedules = data?.items ?? []
   463→
   464→  const { data: classesData } = useQuery({
   465→    queryKey: qk.classes(),
   466→    queryFn: () => api<{ items: ClassOption[] }>('/academics/classes'),
   467→    enabled: open,
   468→  })
   469→  const classes = useMemo(() => (classesData?.items ?? []).slice().sort((a, b) => a.level - b.level), [classesData])
   470→
   471→  const { data: csData, isLoading: csLoading } = useQuery({
   472→    queryKey: qk.classSubjects(classId === 'NONE' ? undefined : classId),
   473→    queryFn: () => api<{ items: ClassSubjectOption[] }>('/academics/class-subjects', { query: { classId } }),
   474→    enabled: open && formOpen && classId !== 'NONE',
   475→  })
   476→  const classSubjects = csData?.items ?? []
   477→
   478→  const closeForm = () => { setFormOpen(false); setEditTarget(null); setFormError('') }
   479→
   480→  const openAdd = () => {
   481→    setEditTarget(null)
   482→    setClassId('NONE')
   483→    setClassSubjectId('NONE')
   484→    setDate(exam?.startDate ?? '')
   485→    setStartTime('09:00')
   486→    setEndTime('11:00')
   487→    setMaxMarks('100')
   488→    setPassMarks('40')
   489→    setFormError('')
   490→    setFormOpen(true)
   491→  }
   492→
   493→  const openEdit = (s: ScheduleRow) => {
   494→    setEditTarget(s)
   495→    setClassId(s.classSubject.class.id)
   496→    setClassSubjectId(s.classSubject.id)
   497→    setDate(s.date)
   498→    setStartTime(s.startTime)
   499→    setEndTime(s.endTime)
   500→    setMaxMarks(String(s.maxMarks))
   501→    setPassMarks(String(s.passMarks))
   502→    setFormError('')
   503→    setFormOpen(true)
   504→  }
   505→
   506→  const validate = (): string => {
   507→    if (classId === 'NONE' || classSubjectId === 'NONE') return 'Select the class and subject for this paper'
   508→    if (!date) return 'Pick the exam date'
   509→    if (!startTime || !endTime) return 'Set both the start and end time'
   510→    if (startTime >= endTime) return 'End time must be after the start time'
   511→    const max = Number(maxMarks)
   512→    const pass = Number(passMarks)
   513→    if (!Number.isFinite(max) || max < 1 || max > 1000) return 'Max marks must be between 1 and 1000'
   514→    if (!Number.isFinite(pass) || pass < 0 || pass > max) return 'Pass marks must be between 0 and the max marks'
   515→    return ''
   516→  }
   517→
   518→  const save = useMutation({
   519→    mutationFn: () => {
   520→      const body = {
   521→        ...(editTarget ? {} : { examId }),
   522→        classSubjectId,
   523→        date,
   524→        startTime,
   525→        endTime,
   526→        maxMarks: Number(maxMarks),
   527→        passMarks: Number(passMarks),
   528→      }
   529→      return editTarget
   530→        ? api(`/exams/schedules/${editTarget.id}`, { method: 'PATCH', body })
   531→        : api('/exams/schedules', { method: 'POST', body })
   532→    },
   533→    onSuccess: () => {
   534→      void qc.invalidateQueries({ queryKey: ['exams'] })
   535→      onChanged()
   536→      closeForm()
   537→      toast.success(editTarget ? 'Paper updated' : 'Paper scheduled')
   538→    },
   539→    onError: e => setFormError(e instanceof Error ? e.message : 'Save failed'),
   540→  })
   541→
   542→  const remove = useMutation({
   543→    mutationFn: (id: string) => api(`/exams/schedules/${id}`, { method: 'DELETE' }),
   544→    onSuccess: () => {
   545→      void qc.invalidateQueries({ queryKey: ['exams'] })
   546→      onChanged()
   547→      setDeleteTarget(null)
   548→      toast.success('Paper removed from the timetable')
   549→    },
   550→    onError: e => { setDeleteTarget(null); toast.error(e instanceof Error ? e.message : 'Delete failed') },
   551→  })
   552→
   553→  const grouped = useMemo(() => {
   554→    const m = new Map<string, ScheduleRow[]>()
   555→    for (const s of schedules) {
   556→      const list = m.get(s.date) ?? []
   557→      list.push(s)
   558→      m.set(s.date, list)
   559→    }
   560→    return [...m.entries()].sort((a, b) => a[0].localeCompare(b[0]))
   561→  }, [schedules])
   562→
   563→  const classSubject = classSubjects.find(cs => cs.id === classSubjectId)
   564→
   565→  return (
   566→    <>
   567→      <Dialog open={open} onOpenChange={o => { onOpenChange(o); if (!o) closeForm() }}>
   568→        <DialogContent className="sm:max-w-2xl">
   569→          <DialogHeader>
   570→            <DialogTitle>Exam timetable — {exam?.name ?? ''}</DialogTitle>
   571→            <DialogDescription>
   572→              Schedule one paper per class subject with its date, time and marks. Marks entry needs at least one paper.
   573→            </DialogDescription>
   574→          </DialogHeader>
   575→
   576→          <div className="max-h-80 space-y-3 overflow-y-auto pr-1 [scrollbar-width:thin]" aria-label="Exam papers">
   577→            {isLoading ? (
   578→              <div className="flex h-24 items-center justify-center"><Loader2 className="h-5 w-5 animate-spin text-primary" aria-hidden /></div>
   579→            ) : grouped.length === 0 ? (
   580→              <p className="rounded-lg border border-dashed px-3 py-6 text-center text-sm text-muted-foreground">
   581→                No papers yet. Add the first paper to unlock marks entry for this exam.
   582→              </p>
   583→            ) : (
   584→              grouped.map(([day, rows]) => (
   585→                <div key={day} className="space-y-1.5">
   586→                  <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">{fmtDate(day)}</p>
   587→                  <ul className="space-y-1.5" role="list" aria-label={`Papers on ${day}`}>
   588→                    {rows.map(s => (
   589→                      <li key={s.id} className="flex flex-col gap-1.5 rounded-lg border px-3 py-2 sm:flex-row sm:items-center sm:gap-3">
   590→                        <div className="min-w-0 flex-1">
   591→                          <p className="truncate text-sm font-medium">
   592→                            {s.classSubject.class.name} — {s.classSubject.subject.name}
   593→                            {s.classSubject.subject.code ? ` (${s.classSubject.subject.code})` : ''}
   594→                          </p>
   595→                          <p className="flex items-center gap-1 text-xs text-muted-foreground">
   596→                            <Clock className="h-3 w-3" aria-hidden />
   597→                            {s.startTime}–{s.endTime} · Max {s.maxMarks} · Pass {s.passMarks}
   598→                            {s.classSubject.teacher ? ` · ${s.classSubject.teacher.firstName} ${s.classSubject.teacher.lastName}` : ''}
   599→                          </p>
   600→                        </div>
   601→                        {canWrite && (
   602→                          <div className="flex gap-1">
   603→                            <Button variant="ghost" size="icon" aria-label={`Edit ${s.classSubject.subject.name} paper`} onClick={() => openEdit(s)}>
   604→                              <Pencil className="h-4 w-4 text-muted-foreground" aria-hidden />
   605→                            </Button>
   606→                            <Button variant="ghost" size="icon" aria-label={`Remove ${s.classSubject.subject.name} paper`} onClick={() => setDeleteTarget(s)}>
   607→                              <Trash2 className="h-4 w-4 text-muted-foreground" aria-hidden />
   608→                            </Button>
   609→                          </div>
   610→                        )}
   611→                      </li>
   612→                    ))}
   613→                  </ul>
   614→                </div>
   615→              ))
   616→            )}
   617→          </div>
   618→
   619→          {formOpen ? (
   620→            <form
   621→              className="space-y-3 rounded-xl border bg-muted/30 p-3"
   622→              onSubmit={e => { e.preventDefault(); const err = validate(); if (err) { setFormError(err); return } setFormError(''); save.mutate() }}
   623→              noValidate
   624→            >
   625→              <p className="text-sm font-semibold">{editTarget ? `Edit ${editTarget.classSubject.subject.name} paper` : 'Add a paper'}</p>
   626→              <div className="grid gap-3 sm:grid-cols-2">
   627→                <div className="space-y-1.5">
   628→                  <Label>Class</Label>
   629→                  <Select value={classId} onValueChange={v => { setClassId(v); setClassSubjectId('NONE') }}>
   630→                    <SelectTrigger aria-label="Select class for the paper"><SelectValue placeholder="Pick a class" /></SelectTrigger>
   631→                    <SelectContent>
   632→                      {classes.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   633→                    </SelectContent>
   634→                  </Select>
   635→                </div>
   636→                <div className="space-y-1.5">
   637→                  <Label>Subject</Label>
   638→                  <Select value={classSubjectId} onValueChange={setClassSubjectId}>
   639→                    <SelectTrigger aria-label="Select subject for the paper"><SelectValue placeholder={classId === 'NONE' ? 'Pick a class first' : 'Pick a subject'} /></SelectTrigger>
   640→                    <SelectContent>
   641→                      {csLoading ? (
   642→                        <SelectItem value="LOADING" disabled>Loading…</SelectItem>
   643→                      ) : classSubjects.length === 0 ? (
   644→                        <SelectItem value="NONE" disabled>No subjects assigned to this class</SelectItem>
   645→                      ) : (
   646→                        classSubjects.map(cs => (
   647→                          <SelectItem key={cs.id} value={cs.id}>
   648→                            {cs.subject.name}{cs.subject.code ? ` (${cs.subject.code})` : ''}
   649→                          </SelectItem>
   650→                        ))
   651→                      )}
   652→                    </SelectContent>
   653→                  </Select>
   654→                  {classSubject?.teacher && <p className="text-xs text-muted-foreground">Teacher: {classSubject.teacher.fullName}</p>}
   655→                </div>
   656→              </div>
   657→              <div className="grid gap-3 sm:grid-cols-3">
   658→                <div className="space-y-1.5">
   659→                  <Label htmlFor="paper-date">Date</Label>
   660→                  <Input id="paper-date" type="date" value={date} onChange={e => setDate(e.target.value)} />
   661→                </div>
   662→                <div className="space-y-1.5">
   663→                  <Label htmlFor="paper-start">Starts</Label>
   664→                  <Input id="paper-start" type="time" value={startTime} onChange={e => setStartTime(e.target.value)} />
   665→                </div>
   666→                <div className="space-y-1.5">
   667→                  <Label htmlFor="paper-end">Ends</Label>
   668→                  <Input id="paper-end" type="time" value={endTime} onChange={e => setEndTime(e.target.value)} />
   669→                </div>
   670→              </div>
   671→              <div className="grid gap-3 sm:grid-cols-2">
   672→                <div className="space-y-1.5">
   673→                  <Label htmlFor="paper-max">Max marks</Label>
   674→                  <Input id="paper-max" type="number" min={1} max={1000} value={maxMarks} onChange={e => setMaxMarks(e.target.value)} />
   675→                </div>
   676→                <div className="space-y-1.5">
   677→                  <Label htmlFor="paper-pass">Pass marks</Label>
   678→                  <Input id="paper-pass" type="number" min={0} max={1000} value={passMarks} onChange={e => setPassMarks(e.target.value)} />
   679→                </div>
   680→              </div>
   681→              {formError && <FieldError message={formError} />}
   682→              <div className="flex justify-end gap-2">
   683→                <Button type="button" variant="outline" size="sm" onClick={closeForm}>Cancel</Button>
   684→                <Button type="submit" size="sm" disabled={save.isPending}>
   685→                  {save.isPending && <Loader2 className="h-4 w-4 animate-spin" aria-hidden />}
   686→                  {editTarget ? 'Save paper' : 'Schedule paper'}
   687→                </Button>
   688→              </div>
   689→            </form>
   690→          ) : (
   691→            canWrite && (
   692→              <Button onClick={openAdd} className="w-full sm:w-auto">
   693→                <Plus aria-hidden /> Add paper
   694→              </Button>
   695→            )
   696→          )}
   697→        </DialogContent>
   698→      </Dialog>
   699→
   700→      <ConfirmDialog
   701→        open={!!deleteTarget}
   702→        onOpenChange={o => { if (!o) setDeleteTarget(null) }}
   703→        title={`Remove ${deleteTarget?.classSubject.subject.name ?? ''} paper?`}
   704→        description="Papers with scores already entered cannot be removed — clear those scores first."
   705→        confirmLabel="Remove paper"
   706→        destructive
   707→        onConfirm={async () => { if (deleteTarget) await remove.mutateAsync(deleteTarget.id) }}
   708→      />
   709→    </>
   710→  )
   711→}
